```python
def highlight_entities_in_text(text, entities):
    """テキストベースでNERエンティティをハイライト"""
    # HTMLタグを含まないテキストに対して安全な置換処理
    for entity_text, category in sorted(entities, key=lambda x: len(x[0]), reverse=True):
        color = CATEGORY_COLORS.get(category, '#CCCCCC')
        highlighted = f'<span style="background-color:{color}; padding:1px 3px; border-radius:3px;">{entity_text}</span>'
        text = text.replace(entity_text, highlighted)
    return text
```
