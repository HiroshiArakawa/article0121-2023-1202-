    # 二段階カラーシステム
    normal_colors = {
        'TIME_PERIOD': '#F0FFF0',        # 薄緑
        'LAW_REFERENCE': '#FFF2F2',      # 薄赤

        color = highlight_colors.get(entity['category'], '#FFD700')
        border_style = "border: 2px solid #FF6B6B; box-shadow: 0 0 10px rgba(255, 107, 107, 0.5);"
    else:
